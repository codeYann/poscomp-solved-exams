\subsection*{Questão 09}

\begin{questionbox}{09}
  Seja a função $f(x,y) = x^2y + 3y^2$. O gradiente $\nabla f(x,y)$ é:
\end{questionbox}

\begin{solutionbox}
  O gradiente é dado por
  \[
  \nabla f(x,y) = \left( \frac{\partial f}{\partial x}, \frac{\partial f}{\partial y} \right).
  \]

  Calculando as derivadas parciais:

  \[
  \frac{\partial f}{\partial x} = 2xy,
  \]
  \[
  \frac{\partial f}{\partial y} = x^2 + 6y.
  \]

  Logo,
  \[
  \nabla f(x,y) = (2xy,\; x^2 + 6y).
  \]
\end{solutionbox}
