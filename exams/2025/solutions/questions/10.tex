\subsection*{Questão 10}

\begin{questionbox}{10}
  As coordenadas do baricentro de um triângulo $ABC$, considerando $A(2,-5)$, $B(4,0)$ e $C(0,2)$, são:
\end{questionbox}

\begin{solutionbox}
  O baricentro $G$ de um triângulo com vértices $A(x_1,y_1)$, $B(x_2,y_2)$ e $C(x_3,y_3)$ é dado por:
  \[
  G = \left( \frac{x_1 + x_2 + x_3}{3}, \frac{y_1 + y_2 + y_3}{3} \right).
  \]

  Substituindo os valores:
  \[
  G = \left( \frac{2 + 4 + 0}{3}, \frac{-5 + 0 + 2}{3} \right)
  = \left( \frac{6}{3}, \frac{-3}{3} \right)
  = (2, -1).
  \]
\end{solutionbox}
